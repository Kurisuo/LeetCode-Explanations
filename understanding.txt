Notes:

    ---> stoi() str to int 
    ----> .push_back to add elements into vector
    ----> using const / & to prevent creating unneccesary copies 


class Codec {
public:

    // Encodes a list of strings to a single string.
    string encode(vector<string>& strs) {
        string encoded = "";

        if(strs.empty()){ 
            return "";
        }

        //we can add the length of the string at the beginning followed by a #
        //what if the string already contains a hash? 
        // ----> we could use a boolean as a lock, while the first has has been read
        //    -----> skip reading all other hashes as a condition and read as ASCII

        //set delimiters and get lengths
        for(const string& x : strs){ //forces x to be a read-only variable 
            int len = x.length();
            encoded.append(to_string(len) + '#' + x);
          //  x.insert(0, '#'); // _#abcd slow, forces shifts everytime
          // x.insert(0, len); // 4#abcd
        }
        return encoded;
    }


    // Decodes a single string to a list of strings.
    vector<string> decode(string s) {
        vector<string> decoded;
        int i = 0;

        //while loop for manual control
        while (i < s.length()){
            int j = s.find('#', i);
            int len = stoi(s.substr(i, j - i));

            decoded.push_back(s.substr(j + 1, len));
            i = j + len + 1;
        }

        return decoded;
    }
};

##LeetCode 160 Intersection of Two Linked Lists

--> The strategy I used for this one involved iterating until we hit nullptr, in which we would have the pntr switch to point to the head of the other node head and iterate against
---> By doing this we guarentee: (a + b) = intersection

    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {

        auto* a = headA;
        auto* b = headB;


        while ((a != b )) { 
            
            if(a  == nullptr){ 
                a = headB;
            } 
            else{ a = a -> next;}
            
            if(b  == nullptr){ 
                b = headA;
            } 
            else{ b = b -> next;}
        }       
        

        return a;
    }

---> The lambda function I wrote:  
    [](const vector<int>& left, const vector<int>& right){return left[0] < right[0];} 
    Captures TWO vector groups, names left and right. Because we have captured whole groups,
    we can directly access their values as if using a 1D vector, hence left[0] & right[0];
    The lambda tells our sorting function to sort by ascending index 0s for all groups so for example:
    
    before: [2, 50], [1, 20], [23, 50], [18, 23]
    after: [1,20], [2, 50], [18, 23], [23, 50]

--> We can now use these sorted vectors to check the adjacent values. If i - 1 > i, then we cannot attend all
    
    if(intervals[x][0] < intervals[x - 1][1]){ return false; }

    Here we are checking to see if the group to the RIGHT contains an adjacent value LESS THAN the value of the LEFT group.
